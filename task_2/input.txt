#include <stdlib.h>

int main() {
    int counter;
    int limit = 15;

    // Iterate until the counter reaches the limit
    for (counter = 0; counter <= limit; counter = counter + 1) {
        if (counter % 2 == 0) {
            printf("Even");
        } else {
            printf("Odd");
        }
    }

    /*
    Execution finished.
    Terminating program safely.
    */
    return 0;
}
